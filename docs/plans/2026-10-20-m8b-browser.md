# M8b Browser Interface Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** A user runs the whole flow from a browser: create a run, answer each pending item on a typed card, watch phases and diagnostic summaries, recover after a crash, and read a run record of what was asked and decided.

**Architecture:** A persistent run record (pending items as published, decisions as taken, one sequence with events) backs server-rendered Jinja2 pages. The event stream only says what changed; `run.js` fetches the fragments that changed, keeps two cursors, resyncs on gaps, protects edits, freezes submissions, and handles an ended session. HTMX submits the cards, and the server's idempotent commands make retries safe.

**Tech Stack:** Python 3.12, FastAPI, Jinja2, htmx 2.0.10 (vendored), plain browser JavaScript, Playwright for Python (async API, `pytest-playwright-asyncio`), Chromium.

**Design:** [2026-10-19-m8b-browser-design.md](2026-10-19-m8b-browser-design.md), on top of [2026-10-19-m8-web-ui-design.md](2026-10-19-m8-web-ui-design.md) (M8a). Read M8b sections 3–8 before Task 1.

### Task 2: The run record: `pendings`, one sequence, the trimming watermark, masking

Pending items are kept as published and decisions as taken, append-only, in the same transactions that publish and consume them (design section 4). They take their `seq` from the run's event counter without being events. Events are now trimmed by count, and `events_pruned_through_seq` records what was deleted, so a gap is known from the watermark rather than from holes in the sequence. `missing_env` values are masked in both copies. A migration marks runs from before as `partial`. `RunService.record` and `GET /api/runs/{id}/record` serve the merged record.

```python
async def entries(h, run_id, user="alice"):
    out, after = [], 0
    while True:
        rec = await h.service.record(user, run_id, after, limit=3)
        out += rec["entries"]
        after = rec["after"]
        if not rec["more"]:
            return out, rec


def kinds(es):
    return [(e["type"], e.get("kind")) for e in es if e["source"] == "pending"]


def test_secret_values_are_masked_everywhere_but_names_and_actions_stay():
    payload = {
        "gap": SECRET_GAP,
        "question": "order-service needs DB_PASSWORD.",
        "options": ["hunter2"],
        "default": "hunter2",
        "error": "hunter2 is too short",

    masked = mask_published("question", payload)
    assert "hunter2" not in json.dumps(masked)
    assert masked["gap"]["key"] == "DB_PASSWORD" and masked["default"] == MASK
    assert masked["options"] == [MASK] and masked["error"] == MASK
    assert decision_record("question", payload, {"answer": "s3cret"}) == {
        "action": "answer",
        "answer": MASK,

    assert decision_record("question", payload, {"answer": None}) == {"action": "default"}
    assert decision_record("question", payload, {"action": "abort"}) == {"action": "abort"}
    plain = {"gap": {"kind": "unsupported_version", "key": None}, "default": "8.4"}

def test_one_oversized_entry_is_a_page_of_its_own():

    assert more

async def test_questions_are_published_and_decided_with_secrets_masked(store, catalog):
    h = Harness(store, SimCluster(), catalog)
    run_id = (await h.create(spec=ORDERS_WITHOUT_ENV)).body["run_id"]
    await answer(h, run_id, {"answer": " "}, command_id="a1")  # rejected: re-asked with an error
```

```python
"""The run record: what each pending item showed and what was decided on it, merged with
the run's kept events in one sequence.

The record keeps its own copy of pending items, so it must not become a second, permanent
home for secrets. One masking rule covers published content and decisions alike: every
value of a missing_env question (the answer, the default, options, error text, nested
copies) is masked, as is any question whose key looks like a secret. Variable names and what
the user did ("answered", "took the default", "aborted") stay visible.
"""

import json
import re
from typing import Any

MASK = {"masked": True}
SECRET = re.compile(r"PASSWORD|PASSWD|SECRET|TOKEN|KEY|CREDENTIAL", re.IGNORECASE)


def sensitive(kind: str, payload: dict[str, Any]) -> bool:
    if kind != "question":
        return False
    gap = payload.get("gap") or {}
    return gap.get("kind") == "missing_env" or bool(SECRET.search(gap.get("key") or ""))


def _mask_values(d: dict[str, Any]) -> dict[str, Any]:
    out = dict(d)
    if out.get("default") is not None:
        out["default"] = MASK
    if out.get("options"):
        out["options"] = [MASK for _ in out["options"]]
    if out.get("error"):
        out["error"] = MASK
    return out


def mask_published(kind: str, payload: dict[str, Any]) -> dict[str, Any]:
    """A pending item's payload as the record keeps it."""
    if not sensitive(kind, payload):
        return payload
    masked = _mask_values(payload)
    if isinstance(payload.get("gap"), dict):
        masked["gap"] = _mask_values(payload["gap"])
    return masked


def decision_record(kind: str, payload: dict[str, Any], decision: dict[str, Any]) -> dict:
    """A validated decision as the record keeps it: the action, and values unless masked."""
    if decision.get("action") == "abort":
        return {"action": "abort"}
    if kind == "question":
        if decision.get("answer") is None:
            return {"action": "default"}
        answer = MASK if sensitive(kind, payload) else decision["answer"]
        return {"action": "answer", "answer": answer}
    if kind == "save_descriptors":
        return {"action": "saved"}  # a claim the read-back checks
    return dict(decision)


def size(entry: dict[str, Any]) -> int:
    return len(json.dumps(entry))


def page(
    pendings: list[dict[str, Any]],
    events: list[dict[str, Any]],
    limit: int,
    max_bytes: int,
) -> tuple[list[dict[str, Any]], bool]:
    """Merge rows by seq into one page; True if more rows follow. One entry over max_bytes
    is a page of its own with its content replaced by a marker, so the cursor always moves."""
    merged = sorted([*pendings, *events], key=lambda e: e["seq"])
    out: list[dict[str, Any]] = []
    total = 0
    for entry in merged[:limit]:
        n = size(entry)
        if not out and n > max_bytes:
            body = "decision" if "decision" in entry else "payload"
            out.append({**entry, body: {"truncated": True, "bytes": n}, "truncated": True})
            break
        if out and total + n > max_bytes:
            break
        out.append(entry)
        total += n
    return out, len(merged) > len(out)
```

```python
    """Each operation's effect, applied in order to a copy of compose, as a person approving
    the patch should see it; None when the patch does not apply, so nothing is guessed.

    before/after are {"present": bool, "value": ...}. An add into an array inserts (nothing
    was there before); a remove leaves nothing at that path (later elements move up, they
    are not the "after"); a test changes nothing and is shown as a check."""
    doc = copy.deepcopy(compose)
    steps = []
    for op in patch:
        path = op.get("path", "")
        inserts = op.get("op") == "add" and _parent_is_list(doc, path)
        before = {"present": False} if inserts else _at(doc, path)
        try:
            doc = apply(doc, [op])
        except (jsonpatch.JsonPatchException, JsonPointerException):
            return None
        if op["op"] == "remove":
            after = {"present": False}
        elif op["op"] == "test":
            after = before
        elif inserts and path.endswith("/-"):
            after = {"present": True, "value": copy.deepcopy(op.get("value"))}
        else:
            after = _at(doc, path)
        steps.append(
            {"op": op["op"], "path": path, "before": before, "after": after, "insert": inserts}
        )
    return steps
```

```python
"""Request checks shared by the JSON API and the HTML pages: the session cookie, the exact
Origin, the CSRF header, cookie attributes, and the client address used for rate limits.
Proxy headers are never consulted."""

from fastapi import Request
from fastapi.responses import Response

from cagent.auth import Session, Sessions

SESSION_COOKIE = "cagent_session"
PRE_COOKIE = "cagent_pre"


class Denied(Exception):
    def __init__(self, status: int, code: str):
        self.status, self.code = status, code


class Access:
    def __init__(self, sessions: Sessions, origin: str, secure_cookies: bool):
        self.sessions, self.origin, self.secure = sessions, origin, secure_cookies

    def cookie(self, response: Response, name: str, value: str, max_age: float) -> None:
        response.set_cookie(
            name,
            value,
            max_age=int(max_age),
            httponly=True,
            samesite="strict",
            secure=self.secure,
            path="/",
        )

    def same_origin(self, request: Request) -> None:
        if request.headers.get("origin") != self.origin:  # missing and "null" never match
            raise Denied(403, "bad_origin")

    def session(self, request: Request) -> Session | None:
        """The live session, without counting the request as use."""
        return self.sessions.get(request.cookies.get(SESSION_COOKIE))

    def require(self, request: Request) -> Session:
        s = self.session(request)
        if s is None:
            raise Denied(401, "login_required")
        return s

    def changing(self, request: Request) -> Session:
        """A state-changing request: session, exact Origin, CSRF header. Counts as use."""
        s = self.require(request)
        self.same_origin(request)
        if not Sessions.csrf_ok(s, request.headers.get("x-csrf-token")):
            raise Denied(403, "bad_csrf")
        self.sessions.touch(s)
        return s

    @staticmethod
    def client(request: Request) -> str:
        return request.client.host if request.client else "unknown"
```

```python
"""The browser interface: server-rendered pages and fragments over the same RunService and
access rules as the JSON API.

All HTML comes from Jinja2 templates with autoescaping on; user and model text is never
marked safe, never reaches a script or an event handler, and never chooses an hx-* attribute,
a template, or a link. A strict Content-Security-Policy allows only this origin's scripts and
styles; HTMX is configured through a meta tag before it starts (no eval, no script tags, no
injected styles, no history cache). Requests the page makes by itself never count as use of
the session; only what the user does refreshes it.
"""

import json
import time
import uuid
from pathlib import Path
from typing import Any
from urllib.parse import parse_qs

import yaml
from fastapi import FastAPI, Request
from fastapi.responses import HTMLResponse, RedirectResponse, Response
from fastapi.staticfiles import StaticFiles

from cagent.access import PRE_COOKIE, SESSION_COOKIE, Access, Denied
from cagent.auth import LoginError, RateLimited
from cagent.runs import RunError, RunService

STATIC = Path(__file__).parent / "static"
HTMX_VERSION = "2.0.10"

CSP = (
    "default-src 'self'; script-src 'self'; style-src 'self'; connect-src 'self'; "
    "img-src 'self'; object-src 'none'; base-uri 'none'; frame-ancestors 'none'; "
    "form-action 'self'"
)

# other errors are not swapped (run.js handles them).

    "allowEval": False,
    "allowScriptTags": False,
    "includeIndicatorStyles": False,
    "historyEnabled": False,
    "historyCacheSize": 0,
    "selfRequestsOnly": True,
    "responseHandling": [
        {"code": "204", "swap": False},
        {"code": "[23]..", "swap": True},
        {"code": "409", "swap": True, "error": False},
        {"code": "422", "swap": True, "error": False},
        {"code": "[45]..", "swap": False, "error": True},

    undefined=StrictUndefined,

    time.strftime("%Y-%m-%d %H:%M:%S UTC", time.gmtime(t)) if t is not None else "—"

def render(name: str, status: int = 200, **context: Any) -> HTMLResponse:
    html = templates.get_template(name).render(htmx_config=json.dumps(HTMX_CONFIG), **context)
    return HTMLResponse(html, status_code=status)


async def form(request: Request) -> dict[str, list[str]]:
    return parse_qs((await request.body()).decode("utf-8", "replace"), keep_blank_values=True)


def first(fields: dict[str, list[str]], name: str) -> str:
    return (fields.get(name) or [""])[0]


class SecurityHeaders:
    """Adds the security headers to every response as it starts. A plain ASGI middleware:
    it never wraps a response body, so event streams and their disconnects pass untouched."""

    def __init__(self, app: Any):
        self.app = app

    async def __call__(self, scope: Any, receive: Any, send: Any) -> None:
        if scope["type"] != "http":
            await self.app(scope, receive, send)
            return
        static = scope["path"].startswith("/static/")

        async def send_with_headers(message: dict[str, Any]) -> None:
            if message["type"] == "http.response.start":
                headers = [
                    (k, v)
                    for k, v in message.get("headers", [])
                    if k.lower() not in (b"cache-control",) or static
                ]
                headers += [
                    (b"content-security-policy", CSP.encode()),
                    (b"x-content-type-options", b"nosniff"),
                    (b"referrer-policy", b"no-referrer"),
                ]
                if not static:
                    headers.append((b"cache-control", b"no-store"))
                message = {**message, "headers": headers}
            await send(message)

        await self.app(scope, receive, send_with_headers)


def install(app: FastAPI, service: RunService, access: Access) -> None:
    sessions = access.sessions
    app.mount("/static", StaticFiles(directory=STATIC), name="static")

    app.add_middleware(SecurityHeaders)

    def login_page(request: Request, status: int = 200, error: str | None = None) -> Response:
        """The login form with a new pre-session: a failed login used up the old one."""
        try:
            pre = sessions.new_pre(access.client(request))
        except RateLimited:
            return render("login.html", 429, csrf=None, error="Too many attempts. Wait a minute.")
        response = render("login.html", status, csrf=pre.csrf, error=error)
        access.cookie(response, PRE_COOKIE, pre.id, sessions.pre_ttl)
        return response

    def page_session(request: Request):
        """The session for a page the user opened (counts as use), or None."""
        s = access.session(request)
        if s is not None:
            sessions.touch(s)
        return s

    @app.get("/login")
    async def login_get(request: Request) -> Response:
        return login_page(request)

    @app.post("/login")
    async def login_post(request: Request) -> Response:
        try:
            access.same_origin(request)
        except Denied:
            return login_page(request, 403, "This request did not come from the login page.")
        fields = await form(request)
        try:
            s = sessions.login(
                request.cookies.get(PRE_COOKIE),
                first(fields, "csrf"),
                first(fields, "token"),
                access.client(request),
            )
        except RateLimited:
            return render("login.html", 429, csrf=None, error="Too many attempts. Wait a minute.")
        except LoginError:
            return login_page(request, 401, "Login failed.")
        response = RedirectResponse("/", status_code=303)
        access.cookie(response, SESSION_COOKIE, s.id, sessions.ttl)
        response.delete_cookie(PRE_COOKIE, path="/")
        return response

    @app.get("/")
    async def runs_page(request: Request) -> Response:
        s = page_session(request)
        if s is None:
            return RedirectResponse("/login", status_code=303)
        runs = await service.list_runs(s.user)
        return render("runs.html", user=s.user, csrf=s.csrf, runs=runs)

    @app.get("/runs/new")
    async def new_run_page(request: Request) -> Response:
        s = page_session(request)
        if s is None:
            return RedirectResponse("/login", status_code=303)
        return render(
            "new_run.html",
            user=s.user,
            csrf=s.csrf,
            runners=await service.runners(s.user),
            command_id=uuid.uuid4().hex,  # a retry of this form reuses it
        )

    @app.post("/runs")
    async def create_run(request: Request) -> Response:
        s = access.changing(request)
        fields = await form(request)
        purpose, spec = None, None
        if first(fields, "input") == "spec":
            try:
                spec = yaml.safe_load(first(fields, "spec"))
            except yaml.YAMLError as e:
                return render("form_error.html", 422, error=f"The spec is not valid YAML: {e}")
            if not isinstance(spec, dict):
                return render("form_error.html", 422, error="The spec must be a YAML mapping.")
        else:
            purpose = first(fields, "purpose")
        try:
            acc = await service.create_run(
                s.user,
                first(fields, "command_id"),
                first(fields, "runner"),
                first(fields, "cluster"),
                purpose=purpose,
                spec=spec,
            )
        except RunError as e:
            status = e.status if e.status in (409, 422) else 422
            return render("form_error.html", status, error=CREATE_ERRORS.get(e.code, e.code))
        return Response(status_code=204, headers={"HX-Redirect": f"/runs/{acc.body['run_id']}"})
```

```
// The run page's client: keeps the regions current from the run's event stream.
//
// The server renders every fragment; this module only decides when to fetch one and whether
// a response is still current. It never builds HTML from data. Design: M8b section 5.
(() => {
  "use strict";
  const root = document.getElementById("run");
  if (!root) return;
  const runId = root.dataset.runId;
  const BUSINESS = [
    "run_state", "observation", "observation_failed", "runner",
    "phase_started", "phase_completed", "phase_interrupted", "phase_failed",
  ];
  const AFTER = {
    run_state: ["state", "record", "commands"],
    observation: ["state", "record"],
    observation_failed: ["state", "record"],
    runner: ["state", "record"],
  };

  const s = {
    generation: 0,                         // bumped by resync and expiry: older work is void
    lastEventSeq: Number(root.dataset.eventSeq),
    recordAfter: Number(root.dataset.recordAfter),
    seen: new Set(),                       // record entries already on the page, by seq
    dirty: new Set(),
    inflight: new Set(),
    timer: null,
    backoff: 250,
    expired: false,
    es: null,
  };

  class Void extends Error {}              // a response from an earlier generation
  class Expired extends Error {}

  async function get(url) {
    const generation = s.generation;
    const r = await fetch(url, { credentials: "same-origin", cache: "no-store" });
    if (r.status === 401) { expire(); throw new Expired(); }
    if (!r.ok) throw new Error(`${url}: ${r.status}`);
    const text = await r.text();
    if (generation !== s.generation || s.expired) throw new Void();
    return { r, text };
  }

  function template(text) {
    // Fragments come only from this origin's server, rendered by Jinja with autoescaping;
    // parsing them into an inert <template> runs nothing (and the CSP forbids inline script).
    const t = document.createElement("template");
    t.innerHTML = text;
    return t.content.querySelector("template");
  }

  // --- refresh queue --------------------------------------------------------------------

  function mark(...names) {
    if (s.expired) return;
    names.forEach((n) => s.dirty.add(n));
    schedule(100);
  }

  function schedule(delay) {
    if (s.timer || s.expired) return;
    s.timer = setTimeout(flush, delay);
  }

  function flush() {
    s.timer = null;
    for (const name of [...s.dirty]) {
      if (s.inflight.has(name)) continue;  // fetched again when the current one ends
      s.dirty.delete(name);
      s.inflight.add(name);
      JOBS[name]()
        .then(() => { s.backoff = 250; })
        .catch((e) => {
          if (e instanceof Void || e instanceof Expired) return;
          s.dirty.add(name);               // stays dirty; retried with backoff
          s.backoff = Math.min(s.backoff * 2, 10000);
        })
        .finally(() => {
          s.inflight.delete(name);
          if (s.dirty.size) schedule(s.backoff);
        });
    }
  }

  // --- regions --------------------------------------------------------------------------

  async function refreshState() {
    const { r, text } = await get(`/runs/${runId}/fragments/state`);
    applyState(template(text));
    return Number(r.headers.get("Cagent-Event-Seq"));
  }

  function applyState(tpl) {
    for (const fresh of tpl.content.querySelectorAll("section[data-region]")) {
      const name = fresh.dataset.region;
      const current = document.getElementById(name);
      if (!current || Number(fresh.dataset.seq) < Number(current.dataset.seq)) continue;
      if (name === "pending") applyPending(current, fresh);
      else htmx.swap(current, fresh.outerHTML, { swapStyle: "outerHTML" });
    }
  }

  // A card being edited stays while it is the same item at the same revision. Otherwise
  // the old card is disabled, keeps its input, and the new one comes in beside it.
  function applyPending(current, fresh) {
    const same = current.dataset.pendingId !== "" &&
      current.dataset.pendingId === fresh.dataset.pendingId &&
      current.dataset.revision === fresh.dataset.revision;
    const busy = current.querySelector("form[data-modified], form[data-sending]");
    if (busy && same) { current.dataset.seq = fresh.dataset.seq; return; }
    const old = busy ? busy.closest("article") : null;
    htmx.swap(current, fresh.outerHTML, { swapStyle: "outerHTML" });
    if (old) {
      retire(old, "This item changed since you started. Your input is kept here for reference.");
      document.getElementById("pending").prepend(old);
    }
  }

  function retire(article, message) {
    article.dataset.stale = "true";
    article.querySelectorAll("button, input, select, textarea").forEach((el) => { el.disabled = true; });
    const note = document.createElement("p");
    note.className = "error stale";
    note.textContent = message;
    article.prepend(note);
  }

  async function syncRecord() {
    for (;;) {
      const { r, text } = await get(`/runs/${runId}/fragments/record?after=${s.recordAfter}`);
      const list = root.querySelector("ol.record");
      let open = false;
      for (const li of template(text).content.querySelectorAll("li")) {
        const seq = li.dataset.seq;
        if (seq && s.seen.has(seq)) continue;
        if (seq) s.seen.add(seq);
        list.append(li);
        if (li.querySelector('[data-status="accepted"]')) open = true;
      }
      s.recordAfter = Number(r.headers.get("Cagent-Record-After"));  // after appending
      if (open) mark("commands");
      if (r.headers.get("Cagent-Record-More") !== "true") return;
    }
  }

  async function refreshCommands() {
    const open = root.querySelectorAll('.command[data-status="accepted"]');
    for (const span of open) {
      const { text } = await get(`/runs/${runId}/fragments/command/${span.dataset.commandId}`);
      htmx.swap(span, text, { swapStyle: "outerHTML" });
    }
    if (root.querySelector('.command[data-status="accepted"]')) mark("commands");
  }

  const JOBS = { state: refreshState, record: syncRecord, commands: refreshCommands };

  // --- the event stream -----------------------------------------------------------------

  function connect() {
    const es = new EventSource(`/api/runs/${runId}/events?after=${s.lastEventSeq}`);
    // Control events first: they have no id and may carry the last one over.
    es.addEventListener("resync", () => resync());
    es.addEventListener("session_expired", () => expire());
    for (const type of BUSINESS) {
      es.addEventListener(type, (ev) => {
        const seq = Number(ev.lastEventId);
        if (!(seq > s.lastEventSeq)) return;
        s.lastEventSeq = seq;
        mark(...(AFTER[type] || ["state", "record"]));
      });
    }
    es.onerror = () => {
      // The browser reconnects by itself after a network error; a refused reconnect ends
      // the stream for good, and a state fetch tells why (401 means the session ended).
      if (es.readyState === EventSource.CLOSED && !s.expired) resync();
    };
    s.es = es;
  }

  async function resync() {
    if (s.expired) return;
    if (s.es) s.es.close();
    s.generation += 1;
    s.dirty.clear();
    try {
      s.lastEventSeq = await refreshState();
      await syncRecord();
      await refreshCommands();
      connect();
    } catch (e) {
      if (e instanceof Expired) return;
      setTimeout(resync, (s.backoff = Math.min(s.backoff * 2, 10000)));
    }
  }

  function expire() {
    if (s.expired) return;
    s.expired = true;
    s.generation += 1;
    if (s.es) s.es.close();
    clearTimeout(s.timer);
    document.querySelectorAll("form button").forEach((b) => { b.disabled = true; });
    const banner = document.createElement("div");
    banner.className = "banner";
    banner.setAttribute("role", "alert");
    banner.append("Your session ended. Your input is kept on this page. ");
    const link = document.createElement("a");
    link.href = "/login";
    link.target = "_blank";
```

```markdown
## 10. Implementation findings

- **`Referrer-Policy: no-referrer` breaks the Origin check.** Under it Chromium sends `Origin: null` on the page's own POSTs (form and fetch), and the exact Origin check refuses them. The policy is `same-origin`: same-origin requests carry their origin; nothing leaks to other sites.
```
